#Function Declaration
Wilt Feint @compute_cube(Feint @x);
Wilt Feint @check_odd_or_even(Feint @num);

#Function Definition
 @compute_cube(@num){
    Feint @i;
    Feint @cube = 0;
    Forward(@i = 0; @i < @num; @i++ ){
        @cube += @num * @num;  # Adding num * num, num times
    }
    Rebound @cube;
}

Wilt @check_odd_or_even(@num) {
    Chant @result;
    Dribble (@num % 2 == 0) {
        @result = "Even";
    } Shoot {
        @result = "Odd";
    }
    Rebound @result;
}

Team() {
    Feint @number, @y;
    Chant @x;
    Inbound("Enter a number: ", @number);
    @x = @check_odd_or_even(@number);
    @y = @compute_cube(@number);
    
    Announce("The number is " & @x & "\n");
    Announce("The cube of " @number  " is " @y);
    
}
