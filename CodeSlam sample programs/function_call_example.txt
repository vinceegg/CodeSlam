# Function Call demonstration (1-4 for different operations)
Dunk @functionCall(@choice) {
    Feint @a = 10, @b = 5;
    
    Dribble (@choice == 1){
        Announce("Multiplication: " & @a * @b);
    }Pass (@choice == 2){
        Announce("Division: " & @a / @b);
    }Pass (@choice == 3){
        Announce("Addition: " & @a + @b);
    }Pass (@choice == 4){
        Announce("Subtraction: " & @a - @b);
    }Shoot {
        Announce("Invalid choice");
    }
}

    Feint @choice = 1;  
    @functionCall(@choice);
}
