Dunk @functionCall(@choice) {
    Feint @a = 10, @b = 5;
    
    Dribble (@choice == 1){
        Announce("Multiplication:\n" & @a * @b);
    }Pass (@choice == 2){
        Announce("Division:\n" & @a / @b);
    }Pass (@choice == 3){
        Announce("Addition:\n" & @a + @b);
    }Pass (@choice == 4){
        Announce("Subtraction:\n" & @a - @b);
    }Shoot {
        Announce("Invalid choice\n");
    }
}

Team() {
    Feint @choice = 1;  
    @functionCall(@choice);
}

 }

Dunk @square(@rows) {
    Feint @i, @j;
    Forward(@i=1; @i <= @rows;@i++){
        Forward(@j=1; @j <= @rows; @j++){
            Dribble(@i == 1 OR @i == @rows OR @j==1 OR @j == @rows){
                Announce("* ");
            }Shoot{
                Announce("  ");
            }
        }
        Announce("\n");
    }
}



Dunk @rectangle(@rows, @cols) {

    Forward(@i = 0; @i < @rows; @i++){
        Forward(@j = 0; @j < @cols; @j++){
           Announce("*");
        }
        Announce("\n");
    }
}
