# Shape Drawing in CodeSlam

# Right Half Pyramid
Team() {
    Feint @i = 1;
    Feint @j = 1;
    Feint @rows = 5;
    
    Forward (@i = 1; @i <= @rows; @i++) {
        # Reset j for each row
        @j = 1;
        
        # Print stars equal to the current row number
        Forward (@j = 1; @j <= @i; @j++) {
            Announce("* ");
        }
        
        # Move to the next line after each row
        Announce("\n");
    }
}

# Left Half Pyramid
Team() {
    Feint @i = 1;
    Feint @j = 1;
    Feint @rows = 5;

    Forward (@i = 1; @i <= @rows; @i++) {
        # Print spaces first
        @j = 1;
        Forward (@j = 1; @j <= @rows - @i; @j++) {
            Announce("  ");
        }
        
        # Then print stars
        @j = 1;
        Forward (@j = 1; @j <= @i; @j++) {
            Announce("* ");
        }
        
        # Move to next line
        Announce("\n");
    }  
}

# Right Triangle (Pyramid)
Team() {
    Feint @i = 1;
    Feint @j = 1;
    Feint @rows = 5;
    
    Forward (@i = 1; @i <= @rows; @i++) {
        # Print spaces
        @j = 1;
        Forward (@j = 1; @j <= @rows - @i; @j++) {
            Announce("  ");
        }
        
        # Print stars (2*i-1 stars per row for a pyramid)
        @j = 1;
        Forward (@j = 1; @j <= (2 * @i - 1); @j++) {
            Announce("* ");
        }
        
        # Move to next line
        Announce("\n");
    }  
}
