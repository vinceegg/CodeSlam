# Shape Drawing in CodeSlam

# Right Half Pyramid
Team() {
    Feint @i, @j, @rows = 5;
    Forward (@i = 1; @i <= @rows; @i++) {
        Forward (@j = 1; @j <= @i; @j++) {
            Announce("* ");
        }
        Announce("\n");
    }
}

# Left Half Pyramid
Team() {
    Feint @i, @j, @rows = 5;

    Forward (@i = 1; @i <= @rows; @i++) {
        Forward (@j = @i; @j < @rows; @j++) {
            Announce("  ");
        }
        Forward (@j = 1; @j <= @i; @j++) {
            Announce("* ");
        }
        Announce("\n");
    }  
}

# Right Triangle
Team() {
    Feint @i, @j, @rows = 5;
    
    Forward (@i = 1; @i <= @rows; @i++) {
        Forward (@j = @i; @j < @rows; @j++) {
            Announce("  ");
        }
        Forward (@j = 1; @j <= (2 * @i - 1); @j++) {
            Announce("* ");
        }
        Announce("\n");
    }  
}

# Square
Team(){
    Feint @i, @j, @rows = 5;
    Forward(@i=1; @i <= @rows; @i++){
        Forward(@j=1; @j <= @rows; @j++){
            Dribble(@i == 1 OR @i == @rows OR @j==1 OR @j == @rows){
                Announce("* ");
            }Shoot{
                Announce("  ");
            }
        }
        Announce("\n");
    }
}
