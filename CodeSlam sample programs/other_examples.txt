# Area and Perimeter Calculation in CodeSlam

# Area of a square
Team() {
    Floater @side = 5.0; 
    Floater @area;

    # Calculate area
    @area = @side * @side;

    # Output the result
    Announce("The area of the square with side " & @side & " is: " & @area);
}

# Perimeter of a triangle
Team() {
    Floater @side1 = 3.0;
    Floater @side2 = 4.0;
    Floater @side3 = 5.0;
    Floater @perimeter;

    # Calculate perimeter
    @perimeter = @side1 + @side2 + @side3;

    # Output the result
    Announce("The perimeter of the triangle is: " & @perimeter);
}

# Even or odd checker
Team() {
    Feint @number = 10;  

    # Check even or odd
    Dribble (@number % 2 == 0) {
        Announce("The number " & @number & " is even.");
    } Shoot {
        Announce("The number " & @number & " is odd.");
    }
}

# Leap year checker
Team() {
    Feint @year = 2024;  

    # Check for leap year
    Dribble ((@year % 4 == 0 AND @year % 100 != 0) OR (@year % 400 == 0)) {
        Announce(@year & " is a leap year.");
    } Shoot {
        Announce(@year & " is not a leap year.");
    }
}
